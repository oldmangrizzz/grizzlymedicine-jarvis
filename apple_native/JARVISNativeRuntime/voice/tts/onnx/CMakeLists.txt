# ONNX Runtime — try find_package first, then fall back to FetchContent binary
find_package(onnxruntime QUIET CONFIG)
if(NOT onnxruntime_FOUND)
    # Try brew location (macOS)
    find_path(ORT_INCLUDE_DIR onnxruntime_cxx_api.h
        PATHS
            /opt/homebrew/include/onnxruntime
            /usr/local/include/onnxruntime
            /opt/homebrew/opt/onnxruntime/include
            /usr/local/opt/onnxruntime/include
        NO_DEFAULT_PATH
    )
    find_library(ORT_LIB onnxruntime
        PATHS
            /opt/homebrew/lib
            /usr/local/lib
            /opt/homebrew/opt/onnxruntime/lib
            /usr/local/opt/onnxruntime/lib
        NO_DEFAULT_PATH
    )
    if(ORT_INCLUDE_DIR AND ORT_LIB)
        message(STATUS "Found ONNX Runtime via brew: ${ORT_LIB}")
        add_library(onnxruntime::onnxruntime SHARED IMPORTED)
        set_target_properties(onnxruntime::onnxruntime PROPERTIES
            IMPORTED_LOCATION "${ORT_LIB}"
            INTERFACE_INCLUDE_DIRECTORIES "${ORT_INCLUDE_DIR}")
    else()
        # SHA-pinned 2026-05-24; onnxruntime v1.18.1 prebuilt binaries.
        # SHA256 hashes computed from official GitHub release tarballs.
        #   arm64:  f3356203e9b6f5023168a12db74b1060ab397f8f3ce8f5cb2c2bd9e7f1195b01
        #   x86_64: 938198521ecccd6fca4cadd627a57966de092feb6ea9ecb579437898d04ebad8
        #   linux:  9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6
        include(FetchContent)
        if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64")
            set(ORT_RELEASE_URL
                "https://github.com/microsoft/onnxruntime/releases/download/v1.18.1/onnxruntime-osx-arm64-1.18.1.tgz")
            set(ORT_RELEASE_HASH "SHA256=f3356203e9b6f5023168a12db74b1060ab397f8f3ce8f5cb2c2bd9e7f1195b01")
        elseif(APPLE)
            set(ORT_RELEASE_URL
                "https://github.com/microsoft/onnxruntime/releases/download/v1.18.1/onnxruntime-osx-x86_64-1.18.1.tgz")
            set(ORT_RELEASE_HASH "SHA256=938198521ecccd6fca4cadd627a57966de092feb6ea9ecb579437898d04ebad8")
        else()
            set(ORT_RELEASE_URL
                "https://github.com/microsoft/onnxruntime/releases/download/v1.18.1/onnxruntime-linux-x64-1.18.1.tgz")
            set(ORT_RELEASE_HASH "SHA256=9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
        endif()
        FetchContent_Declare(onnxruntime_bin
            URL                    "${ORT_RELEASE_URL}"
            URL_HASH               "${ORT_RELEASE_HASH}"
            DOWNLOAD_EXTRACT_TIMESTAMP TRUE
        )
        FetchContent_MakeAvailable(onnxruntime_bin)
        set(ORT_ROOT "${onnxruntime_bin_SOURCE_DIR}")
        add_library(onnxruntime::onnxruntime SHARED IMPORTED)
        set_target_properties(onnxruntime::onnxruntime PROPERTIES
            IMPORTED_LOCATION "${ORT_ROOT}/lib/libonnxruntime.dylib"
            INTERFACE_INCLUDE_DIRECTORIES "${ORT_ROOT}/include")
    endif()
endif()

function(add_jarvis_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE
        jarvis_tts_onnx
        Catch2::Catch2WithMain
        nlohmann_json::nlohmann_json
    )
    target_compile_definitions(${name} PRIVATE ${TEST_COMPILE_DEFS})
    add_test(NAME ${name} COMMAND ${name} --reporter console --success)
endfunction()
